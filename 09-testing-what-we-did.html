shared_examples.rb

# Don't decode the QR (that needs a screenshot + decoder) - just guard
# against the likely failure, a blank code. A real QR has a square
# module grid (viewBox) and a long single path; a blank one has neither.
def assert_qr_rendered(svg)
  view_box = find(svg)[:viewBox].to_s.split
  expect(view_box[2]).to eq view_box[3]      # square module grid
  expect(view_box[2].to_i).to be >= 21       # version-1 grid or bigger
  expect(find("#{svg} path")[:d].to_s.length).to be > 500 # not blank
end
  • Three structural assertions, no decoder. Takes a selector — one helper covers every page.
